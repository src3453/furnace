cmake_minimum_required(VERSION 3.16)
project(si22_test LANGUAGES CXX)
add_executable(si22_playback playback.cpp)
target_compile_features(si22_playback PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(si22_playback PRIVATE /UNDEBUG /utf-8)
else()
  target_compile_options(si22_playback PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME si22_playback COMMAND si22_playback)
